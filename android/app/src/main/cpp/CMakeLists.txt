cmake_minimum_required(VERSION 3.22)
project(zoot_guest LANGUAGES C CXX ASM)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(NOT ANDROID_ABI STREQUAL "arm64-v8a")
    message(FATAL_ERROR "This target is the arm64 guest, not the x64 host")
endif()
set(DOBBY_DEBUG OFF CACHE BOOL "" FORCE)
set(DOBBY_BUILD_EXAMPLE OFF CACHE BOOL "" FORCE)
set(DOBBY_BUILD_TEST OFF CACHE BOOL "" FORCE)
set(Plugin.SymbolResolver OFF CACHE BOOL "" FORCE)
set(NearBranch ON CACHE BOOL "" FORCE)
add_subdirectory("${ZOOT_NATIVE_DEPS_DIR}/Dobby" dobby-build EXCLUDE_FROM_ALL)
add_subdirectory("${ZOOT_NATIVE_DEPS_DIR}/BNM" bnm-build)
target_include_directories(BNM PUBLIC "${ZOOT_NATIVE_DEPS_DIR}/Dobby/include")
target_link_libraries(BNM PUBLIC dobby_static log dl)
add_library(akl_core SHARED ak_core.cpp hook_smoke.cpp hook_fixture.S)
target_compile_options(akl_core PRIVATE -Wall -Wextra -fvisibility=hidden)
target_link_libraries(akl_core PRIVATE BNM dobby_static log dl)
add_executable(akl_hook_smoke hook_smoke.cpp hook_fixture.S hook_smoke_main.cpp)
target_link_libraries(akl_hook_smoke PRIVATE dobby_static log dl)
